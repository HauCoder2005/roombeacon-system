import re
import base64
import urllib.parse
def get_query(url):
    qs = urllib.parse.parse_qs(urllib.parse.urlparse(url).query)
    if 'pb' in qs:
        pb_val = qs['pb'][0]
        # !2z is base64 encoded
        m = re.search(r'!2z([a-zA-Z0-9+/=]+)', pb_val)
        if m:
            b64 = m.group(1)
            # pad if necessary
            b64 += '=' * (-len(b64) % 4)
            try:
                decoded = base64.b64decode(b64).decode('utf-8')
                return decoded
            except Exception as e:
                pass
        
        # !2s is URL encoded
        m2 = re.search(r'!2s([^!]+)', pb_val)
        if m2:
            return urllib.parse.unquote_plus(m2.group(1))
    return None
