import re
from urllib.parse import urlparse, parse_qs, unquote_plus
def extract_from_url(url: str):
    if not url:
        return None
        
    parsed = urlparse(url)
    is_google = "google.com" in parsed.netloc and "/maps" in parsed.path
    is_maps_google = "maps.google.com" in parsed.netloc
    
    if not (is_google or is_maps_google):
        return None
        
    qs = parse_qs(parsed.query)
    
    # Case 1: q coordinates or q text
    if 'q' in qs:
        q_val = qs['q'][0]
        coord_match = re.match(r'^(-?\d+\.\d+),(-?\d+\.\d+)$', q_val)
        if coord_match:
            return {"provider":"google_maps_embed", "map_url_raw":url, "latitude":coord_match.group(1), "longitude":coord_match.group(2)}
        return {"provider":"google_maps_embed", "map_url_raw":url, "query_raw":unquote_plus(q_val).replace('+', ' ')}
        
    # extract pb=!2s...
    # The format pb=... contains sections separated by !
    if 'pb' in qs:
        pb_val = qs['pb'][0]
        s_match = re.search(r'!2s([^!]+)', pb_val)
        if s_match:
            # wait, it could be !2s or !1s something. Usually !2z base64 or !2sUrlEncoded
            print(f"pb !2s Match: {s_match.group(1)}")
            
    embed_match = re.search(r'!3d(-?\d+\.\d+)!4d(-?\d+\.\d+)', url)
    if embed_match:
        return {"provider":"google_maps_embed", "map_url_raw":url, "latitude":embed_match.group(1), "longitude":embed_match.group(2)}
            
    embed_match2 = re.search(r'!2d(-?\d+\.\d+)!3d(-?\d+\.\d+)', url)
    if embed_match2:
        return {"provider":"google_maps_embed", "map_url_raw":url, "latitude":embed_match2.group(2), "longitude":embed_match2.group(1)}
            
    path_match = re.search(r'/@(-?\d+\.\d+),(-?\d+\.\d+)', parsed.path)
    if path_match:
        return {"provider":"google_maps_embed", "map_url_raw":url, "latitude":path_match.group(1), "longitude":path_match.group(2)}
    
    if 'll' in qs:
        ll_val = qs['ll'][0]
        coord_match = re.match(r'^(-?\d+\.\d+),(-?\d+\.\d+)$', ll_val)
        if coord_match:
            return {"provider":"google_maps_embed", "map_url_raw":url, "latitude":coord_match.group(1), "longitude":coord_match.group(2)}
                
    return None
